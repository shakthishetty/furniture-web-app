## 🧠 Step 2: Modify the Add Material Modal (React)

In your `AddMaterialModal.tsx` (or similar component inside `/components/admin/assets`),
add a dropdown to select preset material:

```tsx
import { defaultWoodTypes } from "@/server/data/defaultMaterials";
import { useState } from "react";

function AddMaterialModal() {
  const [selectedPreset, setSelectedPreset] = useState<any>(null);
  const [formData, setFormData] = useState({
    name: "",
    description: "",
    priceModifier: 0,
    priceMultiplier: 1.0,
    color: "#000000",
  });

  const handlePresetSelect = (presetName: string) => {
    const preset = defaultWoodTypes.find((m) => m.name === presetName);
    if (preset) {
      setFormData({
        ...formData,
        name: preset.name,
        description: preset.description,
        priceMultiplier: preset.priceMultiplier,
        color: preset.color,
      });
      setSelectedPreset(preset.name);
    }
  };

  return (
    <div>
      <label className="block text-sm font-medium">Select Preset Material</label>
      <select
        className="border rounded p-2 w-full mt-1"
        onChange={(e) => handlePresetSelect(e.target.value)}
        value={selectedPreset || ""}
      >
        <option value="">-- Select Material Preset --</option>
        {defaultWoodTypes.map((m) => (
          <option key={m.name} value={m.name}>
            {m.name}
          </option>
        ))}
      </select>

      {/* Existing form fields below */}
      <input
        type="text"
        placeholder="Enter material name"
        value={formData.name}
        className="border rounded p-2 mt-2 w-full"
      />
      {/* ... your other fields ... */}
    </div>
  );
}
```
